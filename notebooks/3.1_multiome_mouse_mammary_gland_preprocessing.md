``` r
#remove the _umi_Counts.tsv.gz from the UMI RNA cell barcodes so that they match the full length RNA barcodes

rna_seurat_list[["umi"]] <- lapply(rna_seurat_list[["umi"]], function(seurat) {
  colnames(seurat) <- sub("_umi.*", "", colnames(seurat))
  return(seurat)
})


# add metadata and change colnames to BC_epi
exp_to_md <- c(exp1 = "exp1_3", exp2 = "exp1_3", exp3 = "exp1_3",
               exp4_5 = "exp4_5")

rna_seurat_list <- lapply(rna_seurat_list, function(sublist) {
  out <- lapply(names(sublist), function(expname) {
    seurat <- sublist[[expname]]

    md_name <- exp_to_md[expname]
    if (is.na(md_name)) return(seurat)
    md <- metadata[[md_name]]
    if (is.null(md)) return(seurat)

    # keep only cells whose BC_RNA is in metadata (remove cells not from this experiment)
    keep_cells <- intersect(colnames(seurat), rownames(md))
    if (!length(keep_cells)) return(seurat)
    seurat <- subset(seurat, cells = keep_cells)

    # add metadata
    seurat <- AddMetaData(seurat, metadata = md[keep_cells, , drop = FALSE])

    # rename to BC_epi 
    if ("BC_epi" %in% colnames(seurat@meta.data)) {
      newnames <- seurat@meta.data$BC_epi
      ok <- !is.na(newnames) & nzchar(newnames)
      colnames(seurat)[ok] <- newnames[ok]
    }

    seurat
  })
  names(out) <- names(sublist)
  out
})
```

``` r
seurat_list <- lapply(names(fragpaths), function(mark) {
  lapply(names(fragpaths[[mark]]), function(exp_id) {
    
    message(paste0("### Processing ", mark, " / ", exp_id))
    
    fragpath <- fragpaths[[mark]][[exp_id]]
    
    message("### Counting fragments")
    total_counts <- CountFragments(fragpath)
    barcodes <- total_counts$CB

    message("### Creating fragment object")
    frags <- CreateFragmentObject(path = fragpath, cells = barcodes)
    
    message("### Making 50k bin matrix")
    bin50k_kmatrix <- GenomeBinMatrix(
      frags,
      genome = mm10,
      cells = NULL,
      binsize = 50000,
      process_n = 10000,
      sep = c(":", "_"),
      verbose = TRUE)
    
    message("### Making 5k bin matrix")
    bin5k_kmatrix <- GenomeBinMatrix(
      frags,
      genome = mm10,
      cells = NULL,
      binsize = 5000,
      process_n = 30000,
      sep = c(":", "_"),
      verbose = TRUE)
    
    message("### Making peak matrix")
    peak_matrix <- FeatureMatrix(
      frags,
      features = consensus_peaks[[mark]],
      cells = NULL,
      sep = c("-", "-"),
      verbose = TRUE)
  
    message("### Creating chromatin assay")
    chrom_assay <- CreateChromatinAssay(
      counts = bin5k_kmatrix,
      sep = c(":", "_"),
      genome = "mm10",
      fragments = fragpath,
      min.cells = 1,
      min.features = -1)
    
    message("### Creating Seurat object")
    seurat <- CreateSeuratObject(
      counts = chrom_assay,
      assay = paste0("bin_5k_", mark, ""))
    
    message("### Adding other bins")
    
    chrom_assay50 <- CreateChromatinAssay(
    counts = bin50k_kmatrix,
    sep = c(":", "_"),
    genome = "mm10",
    fragments = fragpath,
    min.cells = 1,
    min.features = -1)
    
    assay_name <- paste0("bin_50k_", mark)
    seurat[[assay_name]] <- chrom_assay50
    
    message("### Adding peak assay")
    
    assay_name <- paste0("peaks_", mark)
    seurat[[assay_name]] <- CreateChromatinAssay(
      counts = peak_matrix, genome = "mm10")
    
    message("### Adding gene activity matrix")
    Annotation(seurat) <- annotations_mm10
    
    ga_matrix <- GeneActivity(seurat,
                              assay = paste0("bin_5k_", mark),
                              extend.upstream = 5000,
                              extend.downstream = 0,
                              process_n = 3000)
    assay_name <- paste0("ga_", mark)
    seurat[[assay_name]] <- CreateAssayObject(counts = ga_matrix)
    
    message("### Adding RNA assays")
    seurat <- AddMetaData(seurat, rna_seurat_list[["umi"]][[exp_id]]@meta.data)
    seurat[["RNA_umi"]] <- rna_seurat_list[["umi"]][[exp_id]]@assays[["RNA"]]
    seurat$nCount_RNA_umi <- seurat$nCount_RNA
    seurat$nFeature_RNA_umi <- seurat$nFeature_RNA
    
    seurat <- AddMetaData(seurat, rna_seurat_list[["full_length"]][[exp_id]]@meta.data)
    seurat[["RNA_fl"]] <- rna_seurat_list[["full_length"]][[exp_id]]@assays[["RNA"]]
    seurat$nCount_RNA_fl <- seurat$nCount_RNA
    seurat$nFeature_RNA_fl <- seurat$nFeature_RNA
    
    seurat <- AddMetaData(seurat, total_counts)
    seurat <- FRiP(seurat, assay = paste0("peaks_", mark), total.fragments = "frequency_count")
    seurat$orig.ident <- exp_id
    seurat$mark <- mark
    
    return(seurat)
  }) |> setNames(names(fragpaths[[mark]]))
}) |> setNames(names(fragpaths))


saveRDS(seurat_list, file.path(outputDir_objects, "seurat_list_step0.rds"))
```

``` r
cor <- as.matrix(reducedDim(sce_k4, "Cor"))
hc <- metadata(sce_k4)$hc_cor
dend <- as.dendrogram(hc)
cluster <- colData(sce_k4)$cell_cluster
exp <- colData(sce_k4)$Exp
frags <- colData(sce_k4)$log10_fragments
filtering <- colData(sce_k4)$filtering

# Correlation scores (n cells with whom a cell correlates more than by random chance)
limitC  <- 0.8
scores <- apply(cor, 2, function(x) sum(x > limitC))


cluster_levels <- if (is.factor(cluster)) levels(cluster) else sort(unique(cluster))
cluster_colors <- setNames(mypal[seq_along(cluster_levels)], cluster_levels)

exp_levels <- if (is.factor(exp)) levels(exp) else sort(unique(exp))
exp_colors <- setNames(alt_pal[seq_along(exp_levels)], exp_levels)

filt_levels <- if (is.factor(filtering)) levels(filtering) else sort(unique(filtering))
filt_colors <- setNames(mypal[seq_along(filt_levels)], filt_levels)

frags_col_fun <- colorRamp2(range(frags, na.rm = TRUE), c("white", "black"))
cor_score_col_fun <- colorRamp2(range(scores, na.rm = TRUE), c("white", "firebrick"))

top_annot <- HeatmapAnnotation(
  cell_cluster = cluster,
  exp = exp,
  filtering = filtering,
  log10_frags    = frags,
  correlation_score = scores,
  col = list(cell_cluster = cluster_colors,
             exp = exp_colors,
             filtering = filt_colors,
             log10_frags    = frags_col_fun,
             correlation_score = cor_score_col_fun)
)

p_before_k4 <- Heatmap(
  cor,
  name = "Correlation",
  col = colorRamp2(
    c(min(cor), mean(range(cor)), max(cor)),
    c("navy", "white", "firebrick3")),
  cluster_rows = dend,
  cluster_columns = dend,
  show_row_names = FALSE,
  show_column_names = FALSE,
  top_annotation = top_annot,

)
```
