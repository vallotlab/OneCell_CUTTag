# Cell type scores

``` r
AddCellTypeScore <- function(seurat_obj,
                             marker_table,
                             cell_type,
                             assay = "SCT_fl",
                             layer = "data",
                             score_name = "cell_type_score") {

  marker_genes <- marker_table %>%
    filter(Type == cell_type) %>%
    pull(Gene) %>%
    unique()
  

  expr_matrix <- GetAssayData(seurat_obj, assay = assay, layer = layer)
  
  marker_genes_present <- intersect(marker_genes, rownames(expr_matrix))
  
  if (length(marker_genes_present) == 0) {
    warning(paste0("No marker genes found in assay for cell type: ", cell_type))
    seurat_obj[[score_name]] <- NA
    return(seurat_obj)
  }
  
  score_vector <- Matrix::colMeans(expr_matrix[marker_genes_present, , drop = FALSE])
  
  seurat_obj[[score_name]] <- score_vector
  
  return(seurat_obj)
}
```

``` r
  mat[feats, , drop = FALSE]

fill_view <- function(mat, all_cells) {
  mat <- as.matrix(mat)

  full <- matrix(
    NA_real_,                                   
    nrow = nrow(mat),
    ncol = length(all_cells),
    dimnames = list(rownames(mat), all_cells)
  )

  common <- intersect(colnames(mat), all_cells)
  full[, common] <- as.matrix(mat[, common, drop = FALSE])

  full
}
```

``` r
rename_enrichment_motifs <- function(enrich, id2name, keep_id = TRUE) {

  rename_ids <- function(ids) {
    nm <- unname(id2name[ids])
    nm <- as.character(nm)

    # treat empty / NULL-like as missing
    nm[nm %in% c("", "NA", "NULL")] <- NA_character_

    # fallback to motif ID
    nm[is.na(nm)] <- ids[is.na(nm)]

    if (keep_id) {
      nm <- paste0(nm, " (", ids, ")")
    }

    make.unique(nm)
  }

  # matrices with pathway IDs as rownames
  old_ids <- rownames(enrich$pval)
  new_ids <- rename_ids(old_ids)

  rownames(enrich$feature.sets)   <- new_ids
  rownames(enrich$pval)           <- new_ids
  rownames(enrich$pval.adj)       <- new_ids
  rownames(enrich$set.statistics) <- new_ids

  enrich$sigPathways <- lapply(enrich$sigPathways, function(x) {
    if (length(x) == 0) return(x)
    rename_ids(x)
  })

  enrich
}
```
